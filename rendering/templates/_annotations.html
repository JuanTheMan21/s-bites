{# Shared runtime helper for every annotation overlay (T18C). Computes, once per annotation, the pixel delta between its target element and the container it shares an ancestor chain with (#stage for SINGLE, a block's own #-region for SPLIT_HORIZONTAL) -- baked into a plain tl.set(...) at t=0 by each annotation partial's own script(), which is what keeps an annotation positioned correctly under continuous transforms (camera drift, panel idle-bob) it shares with its target for the rest of the timeline, without re-measuring on every frame. Every layout builds its GSAP timeline paused and never plays it during script evaluation -- tl.fromTo/tl.set calls only record tweens, nothing touches the DOM until something external calls tl.seek(...), later. So getBoundingClientRect() here always observes the same pristine, untransformed CSS layout regardless of when in the script it runs -- no ordering hazard between a block's own script() and an annotation's positioning call. Real bug found by this task's own Phase-0 spike, not by reasoning about it: getBoundingClientRect returns VIEWPORT pixels, which do not equal this composition's own 1920x1080 CSS-pixel space whenever the capture harness renders at a different effective scale (confirmed live -- an un-normalized delta produced hyperframes check's own "escaped_container" finding, an element landing hundreds of pixels outside #root). #root is declared width:1920px, so measuring its own real rendered width gives the exact scale factor in one line -- dividing every measured delta by it converts viewport pixels back into the local CSS-pixel space tl.set(...)/GSAP's x/y transform actually operate in, regardless of what the browser is doing outside this document. #} {% macro annotation_offset_script() %} function hfAnnotationOffset(targetId, containerId) { var target = document.getElementById(targetId); var container = document.getElementById(containerId); if (!target || !container) return { dx: 0, dy: 0, w: 0, h: 0 }; var root = document.getElementById("root"); var scale = root ? root.getBoundingClientRect().width / 1920 : 1; var t = target.getBoundingClientRect(); var c = container.getBoundingClientRect(); return { dx: (t.left - c.left) / scale, dy: (t.top - c.top) / scale, w: t.width / scale, h: t.height / scale, }; } {#- T18E, D121/D122: two real text collisions -- a CHECK caption printed over its own block's headline, and a caption band printed over a dense SEQUENCE_DIAGRAM's last message -- both came from each annotation partial computing its own fixed offset with no idea what container it was in. hfAnnotationPlace replaces that ad hoc math: it measures the annotation element itself (already in the DOM, pre-transform -- same "nothing has played yet" guarantee hfAnnotationOffset relies on), tries the caller's preferred sides against the target in order, and takes the first whose full box stays inside the container's own USABLE box -- below headlineId's bottom edge, above captionLayerId's top edge -- clamping into that box as a last resort rather than ever landing outside it. #} function hfAnnotationPlace(targetId, containerId, annoId, headlineId, captionLayerId, sides) { var off = hfAnnotationOffset(targetId, containerId); var anno = document.getElementById(annoId); var container = document.getElementById(containerId); if (!anno || !container) return { x: off.dx, y: off.dy }; var root = document.getElementById("root"); var scale = root ? root.getBoundingClientRect().width / 1920 : 1; var a = anno.getBoundingClientRect(); var annoW = a.width / scale, annoH = a.height / scale; var c = container.getBoundingClientRect(); var top = 0, bottom = c.height / scale; var headline = headlineId && document.getElementById(headlineId); if (headline) top = Math.max(top, (headline.getBoundingClientRect().bottom - c.top) / scale); var captionLayer = captionLayerId && document.getElementById(captionLayerId); if (captionLayer) { bottom = Math.min(bottom, (captionLayer.getBoundingClientRect().top - c.top) / scale); } var GAP = 12; var candidates = { // "tip": the annotation wrap's own top-left corner (not its centre) lands exactly on the // target's centre -- CURSOR's own design, a glyph whose tip does the pointing. tip: { x: off.dx + off.w / 2, y: off.dy + off.h / 2 }, center: { x: off.dx + off.w / 2 - annoW / 2, y: off.dy + off.h / 2 - annoH / 2 }, above: { x: off.dx + off.w / 2 - annoW / 2, y: off.dy - annoH - GAP }, below: { x: off.dx + off.w / 2 - annoW / 2, y: off.dy + off.h + GAP }, }; var order = sides && sides.length ? sides : ["above", "below", "center"]; for (var i = 0; i < order.length; i++) { var pos = candidates[order[i]]; if (pos.y >= top && pos.y + annoH <= bottom) return pos; } var fallback = candidates[order[0]]; fallback.y = Math.min(Math.max(fallback.y, top), Math.max(top, bottom - annoH)); return fallback; } {% endmacro %}
